async function staffLogin() { const inputVal = document.getElementById('staffEmail').value.trim(); const pass = document.getElementById('staffPass').value.trim(); if(!inputVal || !pass) { alert('الرجاء إدخال اسم المستخدم/الرقم وكلمة المرور!'); return; } try { const { data, error } = await supabaseClient .from('staff') .select('*') .or(`username.eq.${inputVal},cpr.eq.${inputVal},email.eq.${inputVal}`) .maybeSingle(); if (error || !data) { alert('خطأ: اسم المستخدم أو الرقم غير صحيح.'); return; } if (data.password !== pass) { alert('كلمة المرور غير صحيحة!'); return; } if (data.password === '0') { loggedInStaffId = data.id; document.getElementById('loginCard').classList.add('hidden'); document.getElementById('resetCard').classList.remove('hidden'); return; } alert('أهلاً بك يا أستاذ ' + (data.name || '')); localStorage.setItem('current_staff_user', JSON.stringify(data)); window.location.href = 'dashboard.html'; } catch (err) { console.error(err); alert('حدث خطأ أثناء محاولة الاتصال بالسحابة.'); } }
